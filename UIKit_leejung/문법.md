# protocol

- 규격, 규약, 규칙, 청사진, 뼈대
- protocol은 합성이 가능하다(protocol+protocol=protocol)

```swift
protocol UserInfo{
	var name: String{ get set } //get은 필수, set은 선택
	var age: Int{ get set }
	
extension UserInfo { // extension 은 기능확장
	func isAdult() -> Bool { 
		if age > 19 {
			return true
		}
		return false
	}
}

protocol UserScore{
	var score: Int { get set}
}

protocol UserDetailInfo: UserInfo, UserScore{ //protocol 합성
	var ~~
}
	
class Guest: UserDetailInfo{ //채택해서 사용하겠다
	var score: Int = 10
	
	var name: String = "Kim"
	var age: Int = 20
	var height = 180
	
	
}

		self.name = name

	}
}

class VIPMember: UserInfo{
	var name: String = "Lee"
	var age: Int = 10
}

	func present {

}
```
